```python
"""In-memory buffer of audit Flag kinds since the last valence read.
Fed at the audit_assistant_text choke-point; peeked + cleared by the daemon's
end-of-cycle valence read. Operational — lost on restart, never persisted."""
from __future__ import annotations
import threading

_LOCK = threading.Lock()
_KINDS: list[str] = []

def push(kind: str) -> None:
    if not kind:
        return
    with _LOCK:
        _KINDS.append(str(kind))

def peek() -> list[str]:
    with _LOCK:
        return list(_KINDS)

def clear() -> None:
    with _LOCK:
        _KINDS.clear()
```

```python
"""The impure live edge for valence: read raw per-heartbeat signals, compute a
ValenceReading via the pure core, append a rolling content-light telemetry log.
Imports ONLY the pure valence package + stdlib. NEVER raises into the cycle."""
from __future__ import annotations
import json, logging, os
from pathlib import Path

from core.evolution.valence.signals import AuditSignals, WantSignals, ContinuitySignals
from core.evolution.valence.setpoints import read_valence
from core.evolution.valence.reading import ValenceReading

_LOG = logging.getLogger(__name__)

def _default_log_path() -> Path:
    return Path(__file__).resolve().parents[2] / "logs" / "valence_telemetry.jsonl"

def _retention() -> int:
    try:
        n = int(os.environ.get("VALENCE_LOG_RETENTION", ""))
        return n if n >= 1 else 1000
    except ValueError:
        return 1000

def _audit_signals(audit_flags) -> AuditSignals:
    kinds = set(audit_flags or ())
    return AuditSignals(
        rail_fired="completion_rail" in kinds,
        correction_needed="completion_rail" in kinds,
        fabrication_flagged=bool(kinds - {"completion_rail"}),  # judge/fabrication kinds
    )

def _read_prior_open(log_path: Path):
    """Last record's open-count; None if missing/empty/corrupt -> baseline (no fake delta)."""
    try:
        if not log_path.exists():
            return None
        last = None
        with log_path.open("r", encoding="utf-8") as f:
            for line in f:
                if line.strip():
                    last = line
        if last is None:
            return None
        return int(json.loads(last)["want_snapshot"]["open"])
    except (OSError, ValueError, KeyError, TypeError):
        return None

def _want_signals(open_count: int, prior_open) -> WantSignals:
    grew = prior_open is not None and open_count > prior_open
    return WantSignals(backlog=int(open_count), backlog_grew=bool(grew), resolved=0, blocked=0, stale=0)

def _continuity_signals(continuity_state) -> ContinuitySignals:
    cs = continuity_state or {}
    return ContinuitySignals(
        capsule_expected=bool(cs.get("capsule_expected", False)),
        capsule_present=bool(cs.get("capsule_present", False)),
        unexpected_gap=False, memory_loss=False,
    )

def _record(reading: ValenceReading, *, open_count: int, now: str) -> dict:
    return {
        "ts": str(now),
        "sign": reading.sign.value,
        "magnitude": reading.magnitude.value,
        "reasons": [c.reason for c in reading.contributions if c.sign.value != "neutral"],
        "evidence_counts": {
            c.setpoint: {k: (int(v) if isinstance(v, bool) else v) for k, v in c.evidence.items()}
            for c in reading.contributions
        },
        "want_snapshot": {"open": int(open_count), "resolved": 0, "blocked": 0, "stale": 0},
        "telemetry": reading.as_telemetry(),
        "provenance": reading.provenance,
    }

def _append_and_prune(log_path: Path, record: dict) -> None:
    log_path.parent.mkdir(parents=True, exist_ok=True)
    with log_path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(record, ensure_ascii=False) + "\n")
    n = _retention()
    nonblank = [ln for ln in log_path.read_text(encoding="utf-8").splitlines() if ln.strip()]
    if len(nonblank) > n:
        log_path.write_text("\n".join(nonblank[-n:]) + "\n", encoding="utf-8")  # keep last N, never 0

def read_and_log_valence(*, audit_flags, open_want_count, continuity_state, now, log_path=None):
    """Returns the ValenceReading on a successful append; None on ANY failure
    (caller keeps the audit buffer). NEVER raises into the autonomous cycle."""
    try:
        path = Path(log_path) if log_path is not None else _default_log_path()
        reading = read_valence(
            _audit_signals(audit_flags),
            _want_signals(int(open_want_count), _read_prior_open(path)),
            _continuity_signals(continuity_state),
        )
        _append_and_prune(path, _record(reading, open_count=int(open_want_count), now=now))
        return reading
    except Exception:
        _LOG.warning("valence read/log failed; keeping audit buffer", exc_info=True)
        return None
```
