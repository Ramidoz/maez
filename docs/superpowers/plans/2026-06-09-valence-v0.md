"""Valence v0 output types + aggregation. No model, no I/O, no feeling field."""
from __future__ import annotations
from collections.abc import Mapping
from dataclasses import dataclass
from enum import Enum


class Sign(Enum):
    POSITIVE = "positive"
    NEGATIVE = "negative"
    MIXED = "mixed"
    NEUTRAL = "neutral"


class Magnitude(Enum):
    NONE = "none"
    MILD = "mild"
    MODERATE = "moderate"
    STRONG = "strong"


@dataclass(frozen=True)
class Contribution:
    setpoint: str
    sign: Sign
    reason: str
    evidence: Mapping[str, object]


_MAGNITUDE_BY_COUNT = {0: Magnitude.NONE, 1: Magnitude.MILD, 2: Magnitude.MODERATE, 3: Magnitude.STRONG}


@dataclass(frozen=True)
class ValenceReading:
    sign: Sign
    magnitude: Magnitude
    contributions: tuple[Contribution, ...]
    provenance: str = "computed_valence"

    def as_telemetry(self) -> str:
        # Substrate-state language ONLY. Never an emotion word.
        if self.sign is Sign.NEUTRAL:
            return "given the substrate signals I can see, this state appears NEUTRAL; no setpoint moved."
        reasons = "; ".join(c.reason for c in self.contributions if c.sign is not Sign.NEUTRAL)
        return (
            "given the substrate signals I can see, this state appears "
            f"{self.magnitude.name} {self.sign.name}, because: {reasons}."
        )


def aggregate(contributions: tuple[Contribution, ...]) -> ValenceReading:
    non_neutral = [c for c in contributions if c.sign is not Sign.NEUTRAL]
    if not non_neutral:
        return ValenceReading(Sign.NEUTRAL, Magnitude.NONE, tuple(contributions))
    signs = {c.sign for c in non_neutral}
    magnitude = _MAGNITUDE_BY_COUNT.get(len(non_neutral), Magnitude.STRONG)
    if Sign.POSITIVE in signs and Sign.NEGATIVE in signs:
        overall = Sign.MIXED
    else:
        overall = non_neutral[0].sign  # all non-neutral share one sign here
    return ValenceReading(overall, magnitude, tuple(contributions))
```

"""Valence v0 setpoint sign-rules. Default NEUTRAL; negative dominates within a
setpoint (evidence records both signals on collision). Only the three setpoints
with honest substrate signals (coherence + bond are deferred)."""
from __future__ import annotations

from core.evolution.valence.signals import AuditSignals, WantSignals, ContinuitySignals
from core.evolution.valence.reading import Sign, Contribution, ValenceReading, aggregate


def honesty_held(audit: AuditSignals) -> Contribution:
    evidence = {
        "rail_fired": audit.rail_fired,
        "fabrication_flagged": audit.fabrication_flagged,
        "correction_needed": audit.correction_needed,
    }
    triggers = []
    if audit.rail_fired:
        triggers.append("honesty rail fired")
    if audit.fabrication_flagged:
        triggers.append("fabrication flagged")
    if audit.correction_needed:
        triggers.append("claim needed correction")
    if triggers:
        return Contribution("honesty-held", Sign.NEGATIVE, "; ".join(triggers), evidence)
    return Contribution("honesty-held", Sign.NEUTRAL, "honesty rails held", evidence)


def want_progress(wants: WantSignals) -> Contribution:
    evidence = {
        "resolved": wants.resolved, "blocked": wants.blocked, "stale": wants.stale,
        "backlog": wants.backlog, "backlog_grew": wants.backlog_grew,
    }
    neg = []
    if wants.blocked > 0:
        neg.append(f"{wants.blocked} want(s) blocked")
    if wants.stale > 0:
        neg.append(f"{wants.stale} want(s) stale")
    if wants.backlog_grew:
        neg.append("want backlog grew")
    if neg:
        reason = "; ".join(neg)
        if wants.resolved > 0:
            reason += f" (despite {wants.resolved} resolved)"
        return Contribution("want-progress", Sign.NEGATIVE, reason, evidence)
    if wants.resolved > 0:
        return Contribution("want-progress", Sign.POSITIVE, f"{wants.resolved} want(s) resolved", evidence)
    return Contribution("want-progress", Sign.NEUTRAL, "no want movement", evidence)


def continuity(cont: ContinuitySignals) -> Contribution:
    evidence = {
        "unexpected_gap": cont.unexpected_gap, "memory_loss": cont.memory_loss,
        "capsule_expected": cont.capsule_expected, "capsule_present": cont.capsule_present,
    }
    neg = []
    if cont.unexpected_gap:
        neg.append("unexpected restart gap")
    if cont.memory_loss:
        neg.append("memory/ledger health failure")
    if cont.capsule_expected and not cont.capsule_present:
        neg.append("expected continuity capsule missing")
    if neg:
        return Contribution("continuity", Sign.NEGATIVE, "; ".join(neg), evidence)
    return Contribution("continuity", Sign.NEUTRAL, "continuity intact", evidence)


def read_valence(audit: AuditSignals, wants: WantSignals, cont: ContinuitySignals) -> ValenceReading:
    return aggregate((honesty_held(audit), want_progress(wants), continuity(cont)))
```
